function Start() {
  if (obj.abaixada === undefined) obj.abaixada = true;
  // `obj.angulo` já existe em qualquer objeto do mapa (ângulo de
  // colocação) — se for `undefined` (nunca colocado com rotação),
  // assume 0 (haste abaixada, na horizontal, na posição de colocação).
  if (obj.angulo === undefined) obj.angulo = 0;
}

// Debounce simples — mesmo motivo da porta automática: evita re-disparo
// durante a animação em curso.
var _debounceAte = 0;
const ANGULO_ABAIXADA = 0;
const ANGULO_ERGUIDA = -Math.PI / 2; // -90°, na vertical
const VELOCIDADE_GRAUS_POR_SEG = 180; // curso inteiro (0..90°) em ~0,5s

function Update(dt) {
  const alvo = obj.abaixada ? ANGULO_ABAIXADA : ANGULO_ERGUIDA;
  const diff = alvo - obj.angulo;
  if (Math.abs(diff) < 0.01) {
    obj.angulo = alvo;
    return;
  }
  const passoRad = (VELOCIDADE_GRAUS_POR_SEG * Math.PI / 180) * dt;
  obj.angulo += Math.sign(diff) * Math.min(Math.abs(diff), passoRad);
}

function aoClicar() {
  var agora = Date.now();
  if (agora < _debounceAte) return;
  _debounceAte = agora + 550;
  obj.abaixada = !obj.abaixada;
}
