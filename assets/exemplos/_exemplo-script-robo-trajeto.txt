/**
 * Start é chamada antes do primeiro quadro (uma vez, ao iniciar o objeto).
 */
function Start() {
  // Direção do "ida e volta" (só usada no modo 'ida-e-volta') — 1 = indo
  // pra frente na lista, -1 = voltando. O índice do waypoint atual NÃO
  // fica mais numa variável local (`this._wpIndex`) — foi promovido a
  // `obj.propriedades._trajetoIndiceAtual` (ver mais abaixo), tanto pra
  // sobreviver caso a lista de pontos mude de tamanho no meio da execução
  // (editada pelo painel enquanto o robô anda) quanto pro App de
  // Monitoramento de Robôs conseguir ler o progresso de fora.
  this._wpDir = 1;
  if (!obj.propriedades?.trajeto?.length) {
    console.warn(`[Robô "${obj.nome || obj.tipo}"] sem trajeto definido — abra o painel de propriedades deste objeto no Mapa 2D e cadastre os pontos na seção "🗺️ Trajeto do robô" (nenhum movimento acontece até isso ser feito).`);
  }
}

// Velocidade de deslocamento do robô, em metros por segundo.
const VELOCIDADE = 1.0; // 1 m/s

// Distância (metros) considerada "chegou perto o suficiente" do waypoint
// atual pra avançar pro próximo.
const DISTANCIA_CHEGADA = 0.2; // 20cm

/**
 * Update é chamada uma vez por quadro (a cada frame). `dt` = segundos
 * desde o quadro anterior (mesmo parâmetro já usado por outros exemplos de
 * animação do app, ex.: relógio/mesa animados).
 */
function Update(dt) {
  // [13/09/2026 REESCRITO] Lê o trajeto DIRETO do dado estruturado do
  // objeto — `obj.propriedades.trajeto`, editável pelo painel de
  // propriedades (ver comentário grande no topo deste arquivo). Sem
  // trajeto cadastrado, o robô simplesmente fica parado (o aviso já foi
  // mostrado uma vez em Start(), acima).
  const trajeto = obj.propriedades?.trajeto;
  if (!Array.isArray(trajeto) || !trajeto.length) return;

  // Espera parado neste ponto? (campo opcional `esperaMs` de cada
  // waypoint — 0/ausente = não espera). Controlado por tempo REAL
  // (Date.now()), igual à pausa em mesa do robô de copa — é uma duração
  // curta de "parado fazendo alguma coisa", não um horário do prédio.
  if (this._esperaAte && Date.now() < this._esperaAte) return;
  if (this._esperaAte && Date.now() >= this._esperaAte) this._esperaAte = 0;

  // Índice do waypoint atual — persistido em obj.propriedades (não numa
  // variável local), pra sobreviver a mudanças no tamanho da lista feitas
  // pelo painel enquanto o robô já está andando, e pro App de
  // Monitoramento de Robôs conseguir ler ("Ponto X de Y") sem interpretar
  // este código (ver js/view3d.js `_progressoTrajetoRobo`). Sanitiza pra
  // dentro dos limites atuais da lista a cada quadro (defensivo — cobre o
  // caso de o usuário remover pontos deixando o índice salvo fora do
  // range).
  let idx = obj.propriedades._trajetoIndiceAtual;
  if (typeof idx !== 'number' || idx < 0 || idx >= trajeto.length) idx = 0;

  const alvo = trajeto[idx];
  const dx = alvo.x - obj.x;
  const dy = alvo.y - obj.y;
  const dist = Math.hypot(dx, dy);

  if (dist <= DISTANCIA_CHEGADA) {
    // Chegou no waypoint atual — agenda a espera configurada nele (se
    // houver) e avança pro próximo, conforme o modo escolhido no painel
    // (`obj.propriedades.trajetoModo`: 'ciclo' (padrão) ou 'ida-e-volta').
    if (alvo.esperaMs) this._esperaAte = Date.now() + alvo.esperaMs;
    const modo = obj.propriedades.trajetoModo === 'ida-e-volta' ? 'ida-e-volta' : 'ciclo';
    if (modo === 'ida-e-volta') {
      if (trajeto.length === 1) { idx = 0; } // lista de 1 ponto só — evita loop infinito abaixo
      else {
        idx += this._wpDir;
        if (idx >= trajeto.length) { idx = trajeto.length - 2; this._wpDir = -1; }
        else if (idx < 0) { idx = 1; this._wpDir = 1; }
      }
    } else {
      idx = (idx + 1) % trajeto.length; // 'ciclo': volta ao primeiro após o último
    }
    obj.propriedades._trajetoIndiceAtual = idx;
    return; // já chegou este quadro — próximo quadro mira no novo alvo
  }

  // Anda em direção ao waypoint atual, à VELOCIDADE configurada, sem
  // ultrapassar (limita o passo à distância restante).
  const passo = Math.min(VELOCIDADE * dt, dist);
  obj.x += (dx / dist) * passo;
  obj.y += (dy / dist) * passo;

  // Gira obj.angulo pra apontar na direção do movimento — efeito visual
  // simples (sem esqueleto/animação), como pedido. `angulo` é radianos;
  // `Math.atan2(dy, dx)` já devolve o ângulo certo no MESMO sistema usado
  // por objAnguloToRotY (ver comentário grande no topo deste arquivo).
  obj.angulo = Math.atan2(dy, dx);
}
