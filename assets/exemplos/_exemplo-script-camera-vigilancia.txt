 * (assim como no jogo '007 the world is not enough' para Play Station 1)
 * [...] Deve ser possível trocar de câmera ao vivo (e ligar/desligar aquele
 * movimento de 'um lado para o outro', característico das câmeras: vira
 * para um lado em animação, fica um tempinho, depois vira pro outro lado,
 * em movimento de animação, fica um tempinho e reinicia o ciclo). A
 * movimentação da câmera deve ter som [...] O som do motor dela e barulho
 * que faz ao movimentar-se."
 *
 * COMO USAR: crie/edite um objeto do tipo "Câmera" (Ferramentas →
 * Câmeras, mesmo passo a passo de sempre), abra as Propriedades dela e
 * troque o modelo 3D pra "PS1 (vigilância)" no cartão 3D (botão
 * "🎥 Modelo: [...] — trocar" em "Ver em 3D") — ou deixe no modelo padrão,
 * o script abaixo funciona nos dois (só a APARÊNCIA muda; a lente giratória
 * só é DESENHADA de verdade no modelo PS1 — ver engine3d.js `setScene`,
 * bloco "câmeras" — mas o campo `obj.anguloLente` que este script escreve
 * já existe pra qualquer câmera, sem erro nenhum se o modelo visual não
 * desenhar essa parte). Depois adicione um componente Script (mesmo painel
 * de sempre, "🎬 Componentes" → "+ Script") colando este texto.
 *
 * O QUE ESTE SCRIPT FAZ: alterna `obj.anguloLente` (o desvio da cabeça/lente
 * em relação à orientação de MONTAGEM da câmera, `obj.angulo` — ver
 * comentário grande em engine3d.js sobre por que são 2 campos separados)
 * entre um mínimo e um máximo configuráveis logo abaixo, formando o ciclo
 * pedido: vira pra um lado → pausa → vira pro outro lado → pausa → repete.
 * Liga/desliga o SOM sintetizado do motor (ver js/audio.js) exatamente
 * durante as fases de MOVIMENTO, mudo durante as pausas — reflete o "barulho
 * que faz ao movimentar-se" pedido, não um som constante.
 *
 * LIGAR/DESLIGAR AO VIVO: o botão "🔄 Varredura automática" do cartão 3D da
 * câmera (view3d.js `_showCameraCard3D`) alterna `obj.varreduraAtiva`
 * (true/false) — este script CHECA esse campo no início de `Update()`: se
 * `false`, a lente para exatamente onde está (sem pular pra um ângulo fixo,
 * pra não dar um "solavanco" visual ao desligar) e o som do motor é
 * desligado; ligando de novo, o ciclo retoma da fase em que parou.
 *
 * LIMITAÇÃO JÁ CONHECIDA DO PROJETO (mesma de todo script de trajeto de
 * robô, ver _exemplo-script-robo-copa.txt): as pausas usam `Date.now()`
 * (tempo real do navegador/computador) — não existe ainda um "tempo
 * simulado do prédio" que possa ser acelerado/pausado independente do
 * relógio real.
 */

/**
 * Start é chamada antes do primeiro quadro (uma vez, ao iniciar o objeto).
 */
function Start() {
  this._estado = 'indoMax';     // 'indoMax' | 'pausaMax' | 'indoMin' | 'pausaMin'
  this._proximaTrocaEm = 0;     // timestamp (Date.now()) de quando uma pausa termina
  if (obj.anguloLente === undefined || obj.anguloLente === null) obj.anguloLente = 0;
  this._somLigado = false;      // espelha o último `iniciar` mandado pro AudioFX (evita chamadas repetidas à toa)
}

// ---------------------------------------------------------------------
// EDITE AQUI — arco de varredura (em GRAUS, convertidos pra radianos
// automaticamente logo abaixo) e velocidade/pausas do ciclo.
// ---------------------------------------------------------------------
const ANGULO_MIN_GRAUS = -45;   // até onde vira pra um lado (negativo = sentido horário, ver obj.angulo)
const ANGULO_MAX_GRAUS = 45;    // até onde vira pro outro lado
const VELOCIDADE_GRAUS_POR_SEG = 25; // quão rápido a lente gira durante o movimento
const PAUSA_MS = 2500;          // quanto tempo fica parada em cada extremo antes de voltar

const ANGULO_MIN_RAD = ANGULO_MIN_GRAUS * Math.PI / 180;
const ANGULO_MAX_RAD = ANGULO_MAX_GRAUS * Math.PI / 180;
const VELOCIDADE_RAD_POR_SEG = VELOCIDADE_GRAUS_POR_SEG * Math.PI / 180;

/**
 * Liga/desliga o som do motor só quando o estado pedido MUDA — evitar
 * chamar `AudioFX.tocarSomMotor` todo quadro sem necessidade (o próprio
 * js/audio.js já é seguro contra isso, mas fazer a checagem aqui também
 * deixa a intenção do script mais clara pra quem for ler/editar).
 */
function _definirSomMotor(ligar) {
  if (this._somLigado === ligar) return;
  this._somLigado = ligar;
  if (window.AudioFX) window.AudioFX.tocarSomMotor(ligar);
}

/**
 * Update é chamada uma vez por quadro (a cada frame).
 */
function Update(dt) {
  // "ligar/desligar aquele movimento" — pedido verbatim. Com a varredura
  // desligada, a lente fica PARADA onde está (sem som de motor) até religar.
  if (obj.varreduraAtiva === false) {
    _definirSomMotor.call(this, false);
    return;
  }

  if (this._estado === 'pausaMax' || this._estado === 'pausaMin') {
    _definirSomMotor.call(this, false); // parada durante a pausa — sem som
    if (Date.now() >= this._proximaTrocaEm) {
      this._estado = this._estado === 'pausaMax' ? 'indoMin' : 'indoMax';
    }
    return;
  }

  // 'indoMax' ou 'indoMin' — girando de verdade, som ligado.
  _definirSomMotor.call(this, true);
  const alvo = this._estado === 'indoMax' ? ANGULO_MAX_RAD : ANGULO_MIN_RAD;
  const diff = alvo - obj.anguloLente;
  const passo = VELOCIDADE_RAD_POR_SEG * dt;
  if (Math.abs(diff) <= passo) {
    obj.anguloLente = alvo; // chegou exatamente no extremo — evita "ultrapassar" por 1 quadro maior
    this._estado = this._estado === 'indoMax' ? 'pausaMax' : 'pausaMin';
    this._proximaTrocaEm = Date.now() + PAUSA_MS;
    _definirSomMotor.call(this, false); // já chegou — motor desliga na hora, sem esperar o próximo quadro
  } else {
    obj.anguloLente += Math.sign(diff) * passo;
  }
}
