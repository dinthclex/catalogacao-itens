function Start() {
  if (obj.chamando === undefined) obj.chamando = false;
  window.SceneEventBus?.onGlobalEvent?.('elevadorChegou', (dados) => {
    const meuPiso = window.Mapping?.getAndarDaEntidade?.(obj, map) ?? 0;
    if (dados?.piso !== meuPiso || !obj.chamando) return;
    obj.chamando = false;
  });
}

function Update() {
  // Nada por quadro — este exemplo só reage a clique/evento global.
}

function aoClicar() {
  if (obj.chamando) return; // já chamado, esperando a cabine
  obj.chamando = true;
  const piso = window.Mapping?.getAndarDaEntidade?.(obj, map) ?? 0;
  window.SceneEventBus?.emitGlobal?.('chamarElevador', { piso, idBotao: obj.id });
}
